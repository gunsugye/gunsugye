/** 사용자가 확인한 단일 호실 정정. 다른 동명이인/호실에는 적용하지 않습니다. */
function applyDormCorrectionsV315(){
 requireAdminV281_();return lockAndRun_(()=>{
  const fix=DORM_PLAN_DEFAULT_V315_.correction,main=spreadsheet_(),source=livingSourceV272_(),books=[main];
  if(source.id&&source.id!==main.getId())books.push(SpreadsheetApp.openById(source.id));
  const results=[];
  for(const ss of books){
   const props=PropertiesService.getScriptProperties(),marker='DORM_FIX_343_V315_'+ss.getId();
   if(props.getProperty(marker)){results.push({changed:false});continue;}
   const all=dormUnifiedReadV248_(ss,false);
   const candidate=all.list.filter(r=>r.building===fix.building&&r.floor===fix.floor&&[fix.room,fix.fromRoom].includes(r.room));
   if(!candidate.length){results.push({changed:false});continue;}
   const rooms=candidate.filter(r=>r.entries.some(p=>p.name===fix.names[0]&&p.cohort===fix.cohort&&p.company===fix.company)&&r.entries.some(p=>p.name===fix.names[1]));
   if(!rooms.length){results.push({review:true,message:'대상 두 배정자가 같은 호실에서 확인되지 않습니다.'});continue;}
   if(rooms.length!==1||candidate.some(r=>r.room===fix.room&&r.id!==rooms[0].id)){results.push({review:true,message:'343-3호가 중복되어 자동 정정을 보류했습니다.'});continue;}
   const room=rooms[0],kim=room.entries.filter(p=>p.name===fix.names[1]);
   if(kim.length!==1||!['75','76'].includes(kim[0].cohort)||!['3','4'].includes(kim[0].company)||room.entries.length!==2){results.push({review:true,message:'배정 정보가 알려주신 사례와 달라 자동 정정을 보류했습니다.'});continue;}
   if(all.list.some(r=>r.id!==room.id&&r.entries.some(p=>p.status!=='이전편성'&&fix.names.includes(p.name)&&p.cohort===fix.cohort&&p.company===fix.company))){results.push({review:true,message:'같은 기수·중대·이름이 다른 호실에도 있어 자동 정정을 보류했습니다.'});continue;}
   const sh=all.sh,backup=sh.copyTo(ss);backup.setName('생활실_343정정전_'+Date.now());protectLimitedSheetV226_(backup);
   const next=room.entries.map(p=>({...p,cohort:fix.cohort,company:fix.company,status:'수동확인'}));
   dormUnifiedSaveV248_(sh,room,next);
   sh.getRange(room.row,5).setValue(fix.room);sh.getRange(room.row,15).setValue(fix.group);sh.getRange(room.row,10).setValue(iso_());
   // 이후 재통합 시 구형 백업 명단이 잘못된 값을 다시 가져오지 않도록 같은 호실 기록도 정정합니다.
   const legacy=ss.getSheetByName(DORM_ROSTER_V240.TAB);
   if(legacy&&legacy.getLastRow()>1){
    const rows=legacy.getRange(2,1,legacy.getLastRow()-1,Math.min(13,legacy.getMaxColumns())).getDisplayValues();
    rows.forEach((r,i)=>{if(r[0]===room.id&&fix.names.includes(r[7])){legacy.getRange(i+2,5,1,3).setValues([[fix.room,fix.cohort,fix.company]]);legacy.getRange(i+2,10).setValue('수동확인');}});
   }
   props.setProperty(marker,iso_());results.push({changed:true,backup:backup.getName()});
  }
  return {ok:true,changed:results.some(x=>x.changed),review:results.some(x=>x.review),message:results.filter(x=>x.review).map(x=>x.message).join(' / ')};
 });
}
