function dormPlanCompareV318_(rooms,reference,hints=[]){

 // 참고 시트 이름은 단어 전체로 대조하며, 여러 소속에 같은 이름이 있으면 제외하지 않습니다.
 for(const row of hints.filter(r=>['휴학','위탁'].includes(r.status))){
  const tokens=new Set(row.preview.flatMap(c=>String(c.value||'').split(/[\s,，、;:()\[\]{}\/·.\-]+/).filter(Boolean)));
  for(const [name,identities] of names)if(tokens.has(name)&&identities.size===1)excluded.add([...identities][0]);
 }

 return {missingRoom:missing.filter(p=>!isExcluded(p)),excludedCount:missing.filter(isExcluded).length,missingSmart:assigned.filter(p=>!smartKeys.has(key(p)))};
}
